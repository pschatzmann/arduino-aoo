64 // Set mutex on all existing sources

66 stop_requested = false;
67 stopped = false;
68 task.create("aoo_recv", task_stack_size, task_priority, task_core);

74 void end() {
75 // Stop the task first: once it's stopped, nothing else touches sources
76 // concurrently, so AOOReceiver::end() doesn't need the lock here. The
77 // task confirms the stop outside of the lock, so that it is never
78 // suspended or deleted while it holds sources_mutex.
79 if (!task.isEnded()) {
80 stop_requested = true;
81 for (int j = 0; j < 100 && !stopped; j++) delay(1);
82 }
83 task.end();
84 task.remove();
