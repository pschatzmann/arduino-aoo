91 bool begin() override {
92 total_bytes_sent = 0;
93 cfg.redundancy = cfg.redundancy > 0 ? cfg.redundancy : 1;
94 if (p_encoder == nullptr) {
95 LOGE("Encoder not set");
96 return false;
97 }
98 if (p_output == nullptr) {
99 LOGE("Output not set");
100 return false;
101 }
102
103 if (!p_output->begin()) {
104 LOGE("Stream begin failed");
105 return false;
106 }
107
108 if (cfg.sink_targets.empty()) {
109 LOGW("No sink targets configured, defaulting to sink_id=1");
110 }
111
112 // assign random number to stream_id: random(min, max) can not cover the
113 // full int32 range where long is 32 bits, so we combine two 16 bit values
114 if (stream_id == 0) {
115 randomSeed(micros());
116 uint32_t value = ((uint32_t)random(0x10000) << 16) | random(0x10000);
117 stream_id = value == 0 ? 1 : (int32_t)value;
118 }
119
120 // setup resend buffer
121 aoo_out_buffer.resize(cfg.buffer_size);
122
124 encoder_output.source = this;
125 p_encoder->setOutput(encoder_output);
126 if (!p_encoder->begin(audioInfo())) {
127 LOGE("Encoder failed");
128 return false;
129 }
130 LOGI(
131 "AOOSender: stream_id=%d, codec=%s, block_size=%d, cfg.redundancy=%d, "
132 "cfg.codec_delay_samples=%d",
133 stream_id, codecStr(), p_encoder->samplesPerFrame(), cfg.redundancy,
134 cfg.codec_delay_samples);
135
136 // The /start message announces the block size: if the encoder does not
137 // define it, it is the size of the first write, so /start is sent with
138 // the first block
139 start_sent = false;
140 if (p_encoder->samplesPerFrame() > 0) {
141 if (!aoo_send_start()) {
142 LOGE("Failed to send format information");
143 return false;
144 }
145 start_sent = true;
146 }
147 is_streaming = true;
148
149 return AudioOutput::begin();
150 }

154 void end() {
155 if (!is_active && !is_streaming) return;
156 if (is_streaming) {
157 is_streaming = false;
158 forEachSink([&](int sid) {
159 AOOStopSink stop;
160 stop.source_id = cfg.id;
161 stop.sink_id = sid;
162 stop.stream_id = stream_id;
163 stop.last_seq = seq_no - 1;
164 return aoo_send_message(stop);
165 });
166 }
167 if (p_encoder != nullptr) p_encoder->end();
168 AudioOutput::end();

290 void fillStartMessage(AOOStart& aoo_start, int sink) {
291 aoo_start.source_id = cfg.id;
292 aoo_start.sink_id = sink;
293 aoo_start.version = AOO_VERSION;
294 aoo_start.stream_id = stream_id;
295 aoo_start.channels = audioInfo().channels;
296 aoo_start.sample_rate = audioInfo().sample_rate;
297 int enc_block = p_encoder != nullptr ? p_encoder->samplesPerFrame() : 0;
298 if (enc_block > 0) {
299 LOGI("Encoder block size: %d", enc_block);
300 aoo_start.block_size = enc_block;
301 } else {
302 int bytes_per_frame =
303 audioInfo().channels * (audioInfo().bits_per_sample / 8);
304 aoo_start.block_size =
305 bytes_per_frame > 0 ? block_size / bytes_per_frame : block_size;
306 }
307 aoo_start.codec = codecStr();
308 aoo_start.start_time = aoo_ntp_now();
309 aoo_start.codec_delay_samples = cfg.codec_delay_samples;
310 if (strcmp(codecStr(), "pcm") == 0) {
311 codec_ext_data = htonl(pcmBitDepth(audioInfo().bits_per_sample));
312 } else if (strcmp(codecStr(), "opus") == 0) {
313 codec_ext_data = htonl(2049u); // OPUS_APPLICATION_AUDIO
314 }
315 aoo_start.codec_extension = {(uint8_t*)&codec_ext_data,
316 sizeof(codec_ext_data)};
317 }

327 bool aoo_send_data(const uint8_t* audioData, size_t len) {
328 if (!start_sent) {
329 start_sent = true;
330 aoo_send_start();
331 }
332 int32_t current_seq = seq_no++;
333 uint64_t ts = aoo_ntp_now();
334
335 bool ok = forEachSink([&](int sid) {
336 for (int r = 0; r < cfg.redundancy; r++) {
337 if (!aoo_send_block(sid, current_seq, audioData, len, ts, false))
338 return false;
339 }
340 return true;
341 });
342
343 size_t written = aoo_out_buffer.write(current_seq, audioData, len);
344 LOGD("AOOSender: wrote %d bytes of %d to out buffer", (int)written, (int)len);
345 return ok;
346 }

351 bool aoo_send_block(int32_t sink, int32_t seq, const uint8_t* audioData,
352 size_t len, uint64_t ts, bool reply) {
353 int32_t num_frames = (len + cfg.max_frame_size - 1) / cfg.max_frame_size;
354 for (int32_t i = 0; i < num_frames; i++) {
355 int32_t offset = i * cfg.max_frame_size;
356 int32_t frame_len =
357 min((int32_t)(len - offset), (int32_t)cfg.max_frame_size);
358
359 AOOData aoo_data;
360 aoo_data.source_id = cfg.id;
361 aoo_data.sink_id = sink;
362 aoo_data.stream_id = stream_id;
363 aoo_data.seq_no = seq;
364 aoo_data.timestamp = ts;
365 aoo_data.real_sample_rate = audioInfo().sample_rate;
366 aoo_data.channel_onset = cfg.channel_onset;
367 aoo_data.total_size = len;
368 aoo_data.message_data_size = frame_len;
369 aoo_data.total_number_of_frames = num_frames;
370 aoo_data.frame_idx = i;
371 aoo_data.audio_data.data = (uint8_t*)(audioData + offset);
372 aoo_data.audio_data.len = frame_len;
373
374 if (reply) {
375 if (!aoo_reply(aoo_data)) return false;
376 } else {
377 total_bytes_sent += frame_len;
378 if (!aoo_send_message(aoo_data)) return false;
379 }
380 }
381 return true;
382 }

400 bool processRequestStart(OSCData& osc) {
401 TRACED();
402 AOORequestStart req;
403 if (!req.parse(osc.messageData().data, osc.messageData().len)) {
404 LOGE("Failed to parse start request");
405 return false;
406 }
407 LOGI("Start requested by sink %d", req.sink_id);
408 if (!is_active) return false;
409 addSinkTarget(req.sink_id);
410 // otherwise /start is sent with the first block
411 if (!start_sent) return true;
412 AOOStart aoo_start;
413 fillStartMessage(aoo_start, req.sink_id);
414 return aoo_reply(aoo_start);
415 }
416
418 bool processInvite(OSCData& osc) {
419 TRACED();
420 AOOInvite invite;
421 if (!invite.parse(osc.messageData().data, osc.messageData().len)) {
422 LOGE("Failed to parse invite");
423 return false;
424 }
425 LOGI("Invited by sink %d", invite.sink_id);
426 if (!is_active) {
427 AOODecline decline;
428 decline.source_id = cfg.id;
429 decline.sink_id = invite.sink_id;
430 decline.stream_id = invite.stream_id;
431 return aoo_reply(decline);
432 }
433 addSinkTarget(invite.sink_id);
434 // otherwise /start is sent with the first block
435 if (!start_sent) return true;
436 AOOStart aoo_start;
437 fillStartMessage(aoo_start, invite.sink_id);
438 return aoo_reply(aoo_start);
439 }

474 void addSinkTarget(int id) {
475 for (auto& t : cfg.sink_targets) {
476 if (t.id == id) return;
477 }
478 AOOSinkTarget target(id);
479 if (p_output != nullptr) {
480 target.ip = (uint32_t)p_output->senderIP();
481 target.port = p_output->senderPort();
482 }
483 cfg.sink_targets.push_back(target);
484 }

487 void removeSinkTarget(int id) {
488 for (auto it = cfg.sink_targets.begin(); it != cfg.sink_targets.end();
489 ++it) {
490 if (it->id == id) {
491 cfg.sink_targets.erase(it);
492 return;
493 }
494 }
495 }

499 bool processResendRequest(OSCData& osc) {
500 AOOResendData resend;
501 if (!resend.parse(osc.messageData().data, osc.messageData().len)) {
502 LOGE("Failed to parse resend message");
503 return false;
504 }
505 TRACED();
506 for (auto& item : resend.resend_items) {
507 LOGI("Resend - seq: %d frame: %d", item.seq, item.frame);
508 SingleBuffer<uint8_t>* buffer = aoo_out_buffer.get(item.seq);
509 if (buffer != nullptr) {
510 aoo_resend_data(item.seq, resend.sink_id, buffer->data(),
511 buffer->available());
512 } else {
513 LOGE("Resend: Buffer not found for seq %d", item.seq);
514 }
515 }
516 return true;
517 }

521 bool aoo_resend_data(int32_t orig_seq, int32_t target_sink,
522 const uint8_t* audioData, size_t len) {
523 return aoo_send_block(target_sink, orig_seq, audioData, len,
524 aoo_ntp_now(), true);
