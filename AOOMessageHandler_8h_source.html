80 int processMessages() {
81 int count = 0;
82 bool handled = false;
83 while (p_io != nullptr && p_io->available() > 0) {
84 if (!processOneMessage(handled)) break;
85 if (handled) count++;
86 yield(); // Allow other tasks to run in cooperative multitasking environments
87 }
88 if (count > 0) LOGI("Processed %d messages (handled %d)", count, handled);
89 return count;

94 bool processOneMessage(bool& handled) {
95 handled = false;
96 if (p_io == nullptr) return false;
97
98 size_t avail = p_io->available();
99 if (avail == 0) return false;
100
101 if (in_buffer.size() < avail) in_buffer.resize(avail);
102 p_io->setTimeout(1);
103 size_t read = p_io->readBytes(in_buffer.data(), avail);
104 if (read == 0) return false;
105
106 if (aoo_is_binary(in_buffer.data(), read)) {
107 handled = processBinaryMessage(in_buffer.data(), read);
108 } else {
109 handled = processOSCMessage(in_buffer.data(), read);
110 }
111 return true;
112 }

175 bool handlePing(OSCData& osc) {
176 LOGD("Handling ping");
177 if (isSinkAddress()) {
178 // We are a sink, received ping from source → parse AOOPingSink, reply AOOPongSource
179 AOOPingSink ping;
180 if (!ping.parse(osc.data(), osc.size())) {
181 LOGE("Failed to parse ping");
182 return false;
183 }
184 AOOPongSource pong;
185 pong.source_id = ping.source_id;
186 pong.sink_id = aoo_id;
187 pong.t1 = ping.send_time;
188 pong.t2 = aoo_ntp_now();
189 pong.t3 = aoo_ntp_now();
190 retargetToSender();
191 return pong.send(*p_io);
192 } else {
193 // We are a source, received ping from sink → parse AOOPingSource, reply AOOPongSink
194 AOOPingSource ping;
195 if (!ping.parse(osc.data(), osc.size())) {
196 LOGE("Failed to parse ping");
197 return false;
198 }
199 AOOPongSink pong;
200 pong.source_id = aoo_id;
201 pong.sink_id = ping.sink_id;
202 pong.t1 = ping.send_time;
203 pong.t2 = aoo_ntp_now();
204 pong.t3 = aoo_ntp_now();
205 return replyToSender(pong);
206 }

212 if (isSinkAddress()) {
213 // We are a sink, received pong from source
214 AOOPongSink pong;

219 uint64_t t4 = aoo_ntp_now();

221 } else {
222 // We are a source, received pong from sink
223 AOOPongSource pong;
224 if (!pong.parse(osc.data(), osc.size())) {
225 LOGE("Failed to parse pong");
226 return false;

228 uint64_t t4 = aoo_ntp_now();
229 clock_sync.update(pong.t1, pong.t2, pong.t3, t4);
230 }

249 bool processBinaryMessage(const uint8_t* data, size_t len) {
250 TRACED();
251 AOOData aoo_data;
252 if (!aoo_parse_bin_data(data, len, aoo_data)) {
253 stats_.bin_parse_fail++;
254 return false;
255 }
256
257 int msg_id = aoo_data.sink_id;
258 if (aoo_id == 0 && msg_id > 0) {
259 aoo_id = msg_id;
260 LOGI("Auto-assigned ID from binary: %d", aoo_id);
261 }
262 if (msg_id != aoo_id) {
263 stats_.bin_wrong_id++;
264 return false;
265 }
266
267 stats_.binary_data++;
268 if (p_listener == nullptr) return false;
269
270 p_listener->onBinaryData(data, len);
271 p_listener->onData(aoo_data);
272 return true;
273 }

275 bool processOSCMessage(uint8_t* data, size_t len) {
276 TRACED();
277 OSCData osc;
278 osc.setLogActive(log_osc);
279 if (!osc.parse(data, len)) {
280 stats_.parse_errors++;
281 LOGE("Failed to parse OSC message: %d bytes", (int)len);
282 return false;
283 }
284
285 const char* address = osc.getAddress();
286 int id = getIdFromAddress(address);
287 if (id < 0) {
288 LOGD("OSC message ignored (not for %s): %s", address_prefix, address);
289 return false;
290 }
291 if (aoo_id == 0 && id > 0) {
292 aoo_id = id;
293 LOGI("Auto-assigned ID from OSC: %d", aoo_id);
294 }
295 if (id != aoo_id) {
296 LOGD("OSC message for id %d ignored (our id %d)", id, aoo_id);
297 return false;
298 }
299
300 // Ping/pong handled internally
301 if (strstr(address, "/pong") != nullptr) {
302 stats_.osc_pong++;
303 return handlePong(osc);
304 } else if (strstr(address, "/ping") != nullptr) {
305 stats_.osc_ping++;
306 return handlePing(osc);
307 }
308
309 // Everything else delegated to listener
310 if (p_listener == nullptr) return false;
311
312 if (strstr(address, "/start") != nullptr) {
313 stats_.osc_start++;
314 return p_listener->onStart(osc);
315 } else if (strstr(address, "/uninvite") != nullptr) {
316 stats_.osc_uninvite++;
317 return p_listener->onUninvite(osc);
318 } else if (strstr(address, "/invite") != nullptr) {
319 stats_.osc_invite++;
320 return p_listener->onInvite(osc);
321 } else if (strstr(address, "/decline") != nullptr) {
322 stats_.osc_decline++;
323 return p_listener->onDecline(osc);
324 } else if (strstr(address, "/stop") != nullptr) {
325 stats_.osc_stop++;
326 return p_listener->onStop(osc);
327 } else if (strstr(address, "/data") != nullptr) {
328 stats_.osc_data++;
329 // a sink receives audio data, a source receives resend requests
330 if (!isSinkAddress()) return p_listener->onResendRequest(osc);
331 AOOData aoo_data;
332 aoo_data.sink_id = aoo_id;
333 if (!aoo_data.parse(osc.data(), osc.size())) return false;
334 return p_listener->onData(aoo_data);
335 } else {
336 stats_.osc_other++;
337 LOGW("Unsupported address: %s", address);
338 }
339 return false;
340 }
