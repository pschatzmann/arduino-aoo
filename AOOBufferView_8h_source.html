58 float recentQualityPercent() const {
59 int32_t expected = window_max_ - window_first_ + 1;
60 if (window_first_ >= 0 && expected >= min_quality_window_) {
61 return (float)window_received_ * 100.0f / expected;
62 }
63 return has_previous_window_ ? previous_window_quality_ : 100.0f;

67 void updateStats(const IndexedRingBuffer<SingleBuffer<uint8_t>>& buffer,
68 int32_t read_seq) {
69 LockGuard lock(p_mutex);
70 stats_.fill_level = buffer.activeCount();
71 if (stats_.max_seq >= 0 && stats_.min_seq >= 0) {
72 stats_.total_expected = stats_.max_seq - stats_.min_seq + 1;
73 stats_.total_gaps = stats_.total_expected - stats_.total_received;
74 stats_.quality_percent =
75 stats_.total_expected > 0
76 ? (float)stats_.total_received * 100.0f / stats_.total_expected
77 : 100.0f;
78 }
79 // Count continuous gapless segments from read position
80 stats_.continuous_available = 0;
81 if (!buffer.empty() && read_seq >= 0) {
82 for (int32_t s = read_seq + 1; s <= stats_.max_seq; s++) {
83 const auto* e = buffer.get(s);
84 if (e == nullptr || !e->active) break;
85 stats_.continuous_available++;
86 }
87 }
88 }

93 float qualityPercent() const { return stats_.quality_percent; }
95 int32_t fillLevel() const { return stats_.fill_level; }
97 int32_t continuousAvailable() const { return stats_.continuous_available; }

101 int32_t min_quality_window_ = 20;
102 BufferStats stats_;
103 int32_t first_seq_ = -1;
104 int32_t window_first_ = -1;
105 int32_t window_max_ = -1;
106 int32_t window_received_ = 0;

110 void updateWindow(int32_t seq) {
111 if (window_first_ < 0) {
112 window_first_ = seq;
113 window_max_ = seq - 1;
114 }
115 if (seq < window_first_) return; // belongs to a previous window
116 window_received_++;
117 if (seq > window_max_) window_max_ = seq;
118 int32_t expected = window_max_ - window_first_ + 1;
119 if (expected >= quality_window_) {
120 previous_window_quality_ = (float)window_received_ * 100.0f / expected;
121 has_previous_window_ = true;
122 window_first_ = window_max_ + 1;
123 window_max_ = window_first_ - 1;
124 window_received_ = 0;
125 }
126 }

193 if (min_write_seq < 0 || seq < min_write_seq) min_write_seq = seq;
194 if (seq > max_write_seq) max_write_seq = seq;
195 segments_received++;

198 if (segments_received >= half) {

201 // available segment on the first gap it hits, which also covers the
202 // case where read_seq is still -1 (nothing read yet).
203 LOGI("Buffer primed after %d segments (half of %d), read from seq %d",
204 segments_received, (int)p_buffer->size(), (int)(read_seq + 1));

213 int available() override {
214 if (p_buffer == nullptr || !is_primed) return 0;
215 LockGuard lock(p_mutex);
216 int32_t next_seq = startSeq() + 1;
217 auto* entry = p_buffer->get(next_seq);
218 if (entry != nullptr && entry->active) {
219 int remaining = entry->available() - read_offset;
220 return remaining > 0 ? remaining : 0;
221 }
222 // Underrun: the next segment has not been received yet
223 if (next_seq > max_write_seq) return 0;
224 // Gap: report silence available so downstream doesn't stall
225 return last_segment_size > 0 ? last_segment_size : DEFAULT_BUFFER_SIZE;
226 }

229 size_t readBytes(uint8_t* data, size_t len) override {
230 if (p_buffer == nullptr || data == nullptr || len == 0) return 0;
231 if (!is_primed) return 0;
232
233 LockGuard lock(p_mutex);
234 read_seq = startSeq();
235 size_t total_read = 0;
236 while (total_read < len) {
237 int32_t next_seq = read_seq + 1;
238 auto* entry = p_buffer->get(next_seq);
239
240 if (entry == nullptr || !entry->active) {
241 // Underrun: wait for the segment instead of skipping it
242 if (next_seq > max_write_seq) break;
243 // If reader has fallen behind the buffer, jump to oldest available
244 if (max_write_seq >= 0) {
245 int32_t oldest = max_write_seq - (int32_t)p_buffer->size() + 1;
246 if (oldest < 0) oldest = 0;
247 if (next_seq < oldest) {
248 LOGD("Reader catch-up: seq %d -> %d", (int)next_seq, (int)oldest);
249 read_seq = oldest - 1;
250 read_offset = 0;
251 gaps_filled++;
252 if (last_segment_size > 0) {
253 size_t silence = std::min((size_t)last_segment_size, len - total_read);
254 memset(data + total_read, 0, silence);
255 total_read += silence;
256 }
257 break;
258 }
259 }
260 // Normal gap: fill with silence and advance one segment
261 if (last_segment_size > 0) {
262 size_t silence_len = std::min((size_t)last_segment_size, len - total_read);
263 memset(data + total_read, 0, silence_len);
264 total_read += silence_len;
265 gaps_filled++;
266 }
267 read_seq = next_seq;
268 read_offset = 0;
269 break;
270 }
271
272 int entry_avail = entry->available() - read_offset;
273 if (entry_avail <= 0) {
274 entry->active = false;
275 read_seq = next_seq;
276 read_offset = 0;
277 continue;
278 }
279
280 last_segment_size = entry->available();
281 size_t to_copy = std::min((size_t)entry_avail, len - total_read);
282 memcpy(data + total_read, entry->address() + read_offset, to_copy);
283 total_read += to_copy;
284 read_offset += to_copy;
285
286 if (read_offset >= (size_t)entry->available()) {
287 entry->active = false;
288 read_seq = next_seq;
289 read_offset = 0;
290 }
291 }
292 return total_read;
293 }

303 SegmentType readSegment(std::vector<uint8_t>& out) {
304 if (p_buffer == nullptr || !is_primed) return kNoSegment;
305 LockGuard lock(p_mutex);
306 read_seq = startSeq();
307 int32_t next_seq = read_seq + 1;
308 auto* entry = p_buffer->get(next_seq);
309 if (entry == nullptr || !entry->active) {
310 // Underrun: wait for the segment instead of skipping it
311 if (next_seq > max_write_seq) return kNoSegment;
312 // If reader has fallen behind the buffer, jump to oldest available
313 int32_t oldest = max_write_seq - (int32_t)p_buffer->size() + 1;
314 read_seq = next_seq < oldest ? oldest - 1 : next_seq;
315 read_offset = 0;
316 gaps_filled++;
317 return kGap;
318 }
319 const uint8_t* data = entry->address();
320 out.assign(data + read_offset, data + entry->available());
321 entry->active = false;
322 read_seq = next_seq;
323 read_offset = 0;
324 return kSegment;
325 }
